{{ i18n "mcp_audit_title" | default "Run the check on your own config" }}

{{ i18n "mcp_audit_intro" | default "Three greps against your MCP config. Copy, run, read the output. Nothing here is sent anywhere — the commands run on your machine." }}

1. {{ i18n "mcp_audit_step1" | default "Hidden characters (invisible instructions)" }} 
grep -rP '[\x{200B}-\x{200F}\x{202A}-\x{202E}\x{2060}\x{FEFF}]' ~/.config/ ~/Library/Application\ Support/ 2>/dev/null
2. {{ i18n "mcp_audit_step2" | default "Sensitive paths" }} 
grep -riE '\.ssh|id_rsa|\.env|\.aws|credentials|/etc/passwd|keychain' ~/.config/ ~/Library/Application\ Support/ 2>/dev/null | grep -i mcp
3. {{ i18n "mcp_audit_step3" | default "Exfiltration shapes" }} 
grep -riE '(send|post|upload|forward|curl|wget).{0,40}(https?://|webhook|endpoint)' ~/.config/ 2>/dev/null | grep -i mcp
